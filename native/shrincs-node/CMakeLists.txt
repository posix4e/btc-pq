cmake_minimum_required(VERSION 3.22)
project(BTCPQReceiptNode LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(opt BUILD_BITCOIN_BIN BUILD_GUI BUILD_CLI BUILD_TESTS BUILD_TX BUILD_UTIL BUILD_UTIL_CHAINSTATE BUILD_KERNEL_LIB BUILD_KERNEL_TEST ENABLE_WALLET ENABLE_IPC WITH_ZMQ)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_DAEMON ON CACHE BOOL "" FORCE)
add_subdirectory("${CORE_SOURCE}" core)
add_library(pq_receipt SHARED IMPORTED)
set_target_properties(pq_receipt PROPERTIES IMPORTED_LOCATION "${RECEIPT_LIB}")
target_sources(bitcoin_node PRIVATE ../shrincs_receipt.cpp)
target_include_directories(bitcoin_node PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(bitcoin_node PRIVATE pq_receipt)
