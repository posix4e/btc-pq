cmake_minimum_required(VERSION 3.22)
project(BTCPQ_TOY_REGTEST_ONLY LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(opt BUILD_BITCOIN_BIN BUILD_GUI BUILD_CLI BUILD_TESTS BUILD_TX BUILD_UTIL BUILD_UTIL_CHAINSTATE BUILD_KERNEL_LIB BUILD_KERNEL_TEST ENABLE_WALLET ENABLE_IPC WITH_ZMQ)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_DAEMON ON CACHE BOOL "" FORCE)
add_subdirectory("${CORE_SOURCE}" core)
add_executable(btc-pq-toy-verify ../verify.cpp)
target_compile_definitions(btc-pq-toy-verify PRIVATE BTC_PQ_TOY_EXPERIMENT=1)
target_include_directories(btc-pq-toy-verify PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src")
target_link_libraries(btc-pq-toy-verify PRIVATE core_interface bitcoin_consensus bitcoin_util)
add_executable(btc-pq-toy-search ../toy_search.cpp)
target_include_directories(btc-pq-toy-search PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src" "${CORE_SOURCE}/src/secp256k1/include")
target_link_libraries(btc-pq-toy-search PRIVATE core_interface bitcoin_consensus bitcoin_util univalue secp256k1)
