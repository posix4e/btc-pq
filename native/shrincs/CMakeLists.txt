cmake_minimum_required(VERSION 3.22)
project(BTCPQShrincsDemo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
foreach(opt BUILD_BITCOIN_BIN BUILD_DAEMON BUILD_GUI BUILD_CLI BUILD_TESTS BUILD_TX BUILD_UTIL BUILD_UTIL_CHAINSTATE BUILD_KERNEL_LIB BUILD_KERNEL_TEST ENABLE_WALLET ENABLE_IPC WITH_ZMQ)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CORE_SOURCE}" core)
add_library(shrincs_upstream STATIC
  "${SHRINCS_SOURCE}/src/address.cpp" "${SHRINCS_SOURCE}/src/hash.cpp"
  "${SHRINCS_SOURCE}/src/pors_fp.cpp" "${SHRINCS_SOURCE}/src/shrincs.cpp"
  "${SHRINCS_SOURCE}/src/uxmss.cpp" "${SHRINCS_SOURCE}/src/wots_c.cpp"
  "${SHRINCS_SOURCE}/src/xmss.cpp")
target_include_directories(shrincs_upstream PUBLIC "${SHRINCS_SOURCE}/include")
target_compile_definitions(shrincs_upstream PUBLIC SHRINCS_B32)
target_compile_options(shrincs_upstream PRIVATE -Wno-deprecated-declarations)
target_link_libraries(shrincs_upstream PUBLIC OpenSSL::Crypto Threads::Threads)
add_library(shrincs_bridge STATIC ../shrincs_bridge.cpp)
target_link_libraries(shrincs_bridge PRIVATE shrincs_upstream)
target_compile_options(shrincs_bridge PRIVATE -Wno-deprecated-declarations)
add_library(shrincs_bounded STATIC ../shrincs_bounded.cpp)
target_link_libraries(shrincs_bounded PUBLIC OpenSSL::Crypto)
target_compile_options(shrincs_bounded PRIVATE -Wno-deprecated-declarations)
add_executable(btc-pq-shrincs-verify ../shrincs_verify.cpp)
target_include_directories(btc-pq-shrincs-verify PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src")
target_link_libraries(btc-pq-shrincs-verify PRIVATE core_interface bitcoin_consensus bitcoin_util shrincs_bounded)
add_executable(btc-pq-shrincs-tool ../shrincs_tool.cpp)
target_link_libraries(btc-pq-shrincs-tool PRIVATE shrincs_upstream shrincs_bridge shrincs_bounded)
target_compile_options(btc-pq-shrincs-tool PRIVATE -Wno-deprecated-declarations)
add_executable(shrincs-upstream-kat "${SHRINCS_SOURCE}/kat/kat_gen_pass.cpp" "${SHRINCS_SOURCE}/kat/rng.c")
target_include_directories(shrincs-upstream-kat PRIVATE "${SHRINCS_SOURCE}/kat")
target_link_libraries(shrincs-upstream-kat PRIVATE shrincs_upstream)
target_compile_options(shrincs-upstream-kat PRIVATE -Wno-deprecated-declarations)
