cmake_minimum_required(VERSION 3.22)
project(BTCPQReceiptDemo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(opt BUILD_BITCOIN_BIN BUILD_DAEMON BUILD_GUI BUILD_CLI BUILD_TESTS BUILD_TX BUILD_UTIL BUILD_UTIL_CHAINSTATE BUILD_KERNEL_LIB BUILD_KERNEL_TEST ENABLE_WALLET ENABLE_IPC WITH_ZMQ)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CORE_SOURCE}" core)
add_library(pq_receipt SHARED IMPORTED)
set_target_properties(pq_receipt PROPERTIES IMPORTED_LOCATION "${RECEIPT_LIB}")
add_executable(btc-pq-shrincs-receipt ../shrincs_receipt_verify.cpp)
target_include_directories(btc-pq-shrincs-receipt PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src")
target_link_libraries(btc-pq-shrincs-receipt PRIVATE core_interface bitcoin_consensus bitcoin_util univalue pq_receipt)
