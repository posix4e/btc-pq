cmake_minimum_required(VERSION 3.22)
project(BTCPQ LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
foreach(opt BUILD_BITCOIN_BIN BUILD_DAEMON BUILD_GUI BUILD_CLI BUILD_TESTS BUILD_TX BUILD_UTIL BUILD_UTIL_CHAINSTATE BUILD_KERNEL_LIB BUILD_KERNEL_TEST ENABLE_WALLET ENABLE_IPC WITH_ZMQ)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CORE_SOURCE}" core)
add_executable(btc-pq-verify verify.cpp)
target_include_directories(btc-pq-verify PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src")
target_link_libraries(btc-pq-verify PRIVATE core_interface bitcoin_consensus bitcoin_util)
add_executable(btc-pq-phase3-search phase3_search.cpp)
target_include_directories(btc-pq-phase3-search PRIVATE "${CORE_SOURCE}/src" "${CMAKE_BINARY_DIR}/core/src" "${CORE_SOURCE}/src/secp256k1/include")
target_link_libraries(btc-pq-phase3-search PRIVATE core_interface bitcoin_consensus bitcoin_util univalue secp256k1)
